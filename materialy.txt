polecenia sql:
1. logowanie - mysql -uroot -p
2. pokazywanie baz danych - show databases;
3. wybor bazy - use "nazwa_bazy";
4. pokazywanie zawartosci bazy - show tables;
5. pokaywanie zawartosci tabeli - describe "nazwa tabeli"
6. usuwanie tabeli - drop "nazwa_tabeli"
7. tworzenie bazy - create database "nazwa_bazy";
8. tworzenie tabeli - create table "nazwa_tabeli";

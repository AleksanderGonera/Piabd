2. wyswietl uczniow mieszkajacych w lodzi i hobby
SELECT * FROM uczniowie where miejscowosc = 'łódź' and hobby = 'programowanie';

21. wyswietl imie i nazwisko ucznia w jednej kolumnie
SELECT id,
       CONCAT(imie, ' ', nazwisko) AS uczen
FROM uczniowie;
22. wyswietl imiona uczniow wielkimi literami
SELECT id,
       UPPER(imie) AS imie_duzymi_literami,
       nazwisko
FROM uczniowie;
23. wyswietl uczniow urodzonych w czerwcu
SELECT imie, nazwisko, data_urodzenia
FROM uczniowie
WHERE MONTH(data_urodzenia) = 6    funkcja wbudowana które pozwalają wyselekcjonować angielskie nazwy funkcji
ORDER BY data_urodzenia;     domyślnie rosnąco czyli ASC
